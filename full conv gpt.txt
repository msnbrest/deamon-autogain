### Architecture simple

* Capture audio système → **PipeWire** (idéal sous Kubuntu).
* Module **Auto Gain** en temps réel.
* Analyse du niveau RMS/peak.
* **Peak détecté** → multiplicateur baisse rapidement.
* **Pas de peak** → multiplicateur remonte progressivement sur **5 s**.
* Gain unique appliqué aux **2 canaux simultanément** → stéréo conservée.
* Limiteur : **×1 → ×40**.
* Pas de config externe : paramètres directement en constantes dans le code.

### Logique

```text
gain = clamp(gain, 1.0, 40.0)

si peak > seuil:
    gain *= réduction_rapide

sinon:
    gain += montée_lente_sur_5s
```

⚠️ Point important : **×40 peut provoquer du clipping/distorsion**. Pour ton objectif, ajouter un **limiteur/soft-clip très simple en sortie** est fortement recommandé.

## Étape 5 — créer le flux audio stéréo

On va maintenant créer un **flux PipeWire virtuel** en `float32 / stéréo`, avec callback audio. Pour l'instant, il ne fait qu'afficher que des buffers arrivent.

Remplace `autogain.cpp` par :

```cpp id="q7m3kx"
#include <pipewire/pipewire.h>
#include <pipewire/stream.h>
#include <spa/param/audio/format-utils.h>

#include <iostream>

static void on_process(void *data)
{
    pw_stream *stream = static_cast<pw_stream *>(data);

    pw_buffer *b = pw_stream_dequeue_buffer(stream);
    if (!b)
        return;

    std::cout << "Buffer audio reçu\n";

    pw_stream_queue_buffer(stream, b);
}

int main()
{
    pw_init(nullptr, nullptr);

    pw_main_loop *loop = pw_main_loop_new(nullptr);
    pw_context *context =
        pw_context_new(pw_main_loop_get_loop(loop), nullptr, 0);

    pw_core *core = pw_context_connect(context, nullptr, 0);

    if (!core) {
        std::cerr << "Connexion PipeWire impossible\n";
        return 1;
    }

    pw_stream_events events{};
    events.version = PW_VERSION_STREAM_EVENTS;
    events.process = on_process;

    pw_stream *stream = pw_stream_new_simple(
        pw_main_loop_get_loop(loop),
        "Mini Auto Gain",
        nullptr,
        &events,
        nullptr
    );

    spa_audio_info_raw audio{};
    audio.format = SPA_AUDIO_FORMAT_F32;
    audio.rate   = 48000;
    audio.channels = 2;
    audio.position[0] = SPA_AUDIO_CHANNEL_FL;
    audio.position[1] = SPA_AUDIO_CHANNEL_FR;

    auto params = spa_format_audio_raw_build(
        SPA_PARAM_EnumFormat,
        &audio
    );

    pw_stream_connect(
        stream,
        PW_DIRECTION_OUTPUT,
        PW_ID_ANY,
        static_cast<pw_stream_flags>(
            PW_STREAM_FLAG_AUTOCONNECT |
            PW_STREAM_FLAG_MAP_BUFFERS |
            PW_STREAM_FLAG_RT_PROCESS
        ),
        &params,
        1
    );

    std::cout << "Mini Auto Gain démarré\n";

    pw_main_loop_run(loop);

    pw_stream_destroy(stream);
    pw_core_disconnect(core);
    pw_context_destroy(context);
    pw_main_loop_destroy(loop);
    pw_deinit();

    return 0;
}
```

## Étape 6 — vrai filtre stéréo, gain ×1

On crée **2 canaux indépendants dans le code**, mais le gain sera ensuite commun aux deux.

Remplace entièrement `autogain.cpp` :

```cpp
#include <pipewire/pipewire.h>
#include <pipewire/filter.h>

#include <spa/param/audio/format-utils.h>

#include <cstring>
#include <iostream>

struct Data {
    pw_main_loop *loop;
    pw_filter *filter;

    void *in_l;
    void *out_l;

    void *in_r;
    void *out_r;
};

static void on_process(void *userdata, spa_io_position *position)
{
    Data *d = static_cast<Data *>(userdata);

    uint32_t n = position->clock.duration;

    float *in_l  = static_cast<float *>(
        pw_filter_get_dsp_buffer(d->in_l, n));

    float *out_l = static_cast<float *>(
        pw_filter_get_dsp_buffer(d->out_l, n));

    float *in_r  = static_cast<float *>(
        pw_filter_get_dsp_buffer(d->in_r, n));

    float *out_r = static_cast<float *>(
        pw_filter_get_dsp_buffer(d->out_r, n));

    if (!in_l || !out_l || !in_r || !out_r)
        return;

    // Pour l'instant : gain ×1
    std::memcpy(out_l, in_l, n * sizeof(float));
    std::memcpy(out_r, in_r, n * sizeof(float));
}

static const pw_filter_events events = {
    PW_VERSION_FILTER_EVENTS,
    .process = on_process
};

int main()
{
    pw_init(nullptr, nullptr);

    Data d{};

    d.loop = pw_main_loop_new(nullptr);

    d.filter = pw_filter_new_simple(
        pw_main_loop_get_loop(d.loop),
        "Mini Auto Gain",
        pw_properties_new(
            PW_KEY_MEDIA_TYPE, "Audio",
            PW_KEY_MEDIA_CATEGORY, "Filter",
            PW_KEY_MEDIA_ROLE, "DSP",
            nullptr),
        &events,
        &d
    );

    d.in_l = pw_filter_add_port(
        d.filter,
        PW_DIRECTION_INPUT,
        PW_FILTER_PORT_FLAG_MAP_BUFFERS,
        0,
        pw_properties_new(
            PW_KEY_FORMAT_DSP, "32 bit float mono audio",
            PW_KEY_PORT_NAME, "input_FL",
            nullptr),
        nullptr, 0);

    d.out_l = pw_filter_add_port(
        d.filter,
        PW_DIRECTION_OUTPUT,
        PW_FILTER_PORT_FLAG_MAP_BUFFERS,
        0,
        pw_properties_new(
            PW_KEY_FORMAT_DSP, "32 bit float mono audio",
            PW_KEY_PORT_NAME, "output_FL",
            nullptr),
        nullptr, 0);

    d.in_r = pw_filter_add_port(
        d.filter,
        PW_DIRECTION_INPUT,
        PW_FILTER_PORT_FLAG_MAP_BUFFERS,
        0,
        pw_properties_new(
            PW_KEY_FORMAT_DSP, "32 bit float mono audio",
            PW_KEY_PORT_NAME, "input_FR",
            nullptr),
        nullptr, 0);

    d.out_r = pw_filter_add_port(
        d.filter,
        PW_DIRECTION_OUTPUT,
        PW_FILTER_PORT_FLAG_MAP_BUFFERS,
        0,
        pw_properties_new(
            PW_KEY_FORMAT_DSP, "32 bit float mono audio",
            PW_KEY_PORT_NAME, "output_FR",
            nullptr),
        nullptr, 0);

    if (pw_filter_connect(
            d.filter,
            PW_FILTER_FLAG_RT_PROCESS,
            nullptr, 0) < 0) {

        std::cerr << "Erreur connexion filtre\n";
        return 1;
    }

    std::cout << "Mini Auto Gain actif\n";

    pw_main_loop_run(d.loop);

    pw_filter_destroy(d.filter);
    pw_main_loop_destroy(d.loop);
    pw_deinit();
}
```

### Étape 11 — nouveau `autogain.cpp`

Puis mets :

```cpp
#include <pipewire/pipewire.h>
#include <pipewire/filter.h>

#include <iostream>

struct Data {
    pw_main_loop *loop;
    pw_filter *filter;

    void *in_l;
    void *in_r;
    void *out_l;
    void *out_r;
};

static void on_process(void *userdata, spa_io_position *position)
{
    Data *d = static_cast<Data *>(userdata);

    uint32_t n = position->clock.duration;

    float *in_l  = (float *)pw_filter_get_dsp_buffer(d->in_l, n);
    float *in_r  = (float *)pw_filter_get_dsp_buffer(d->in_r, n);
    float *out_l = (float *)pw_filter_get_dsp_buffer(d->out_l, n);
    float *out_r = (float *)pw_filter_get_dsp_buffer(d->out_r, n);

    if (!in_l || !in_r || !out_l || !out_r)
        return;

    // Test : gain ×1
    for (uint32_t i = 0; i < n; ++i) {
        out_l[i] = in_l[i];
        out_r[i] = in_r[i];
    }
}

static const pw_filter_events events = {
    PW_VERSION_FILTER_EVENTS,
    .process = on_process
};

int main()
{
    pw_init(nullptr, nullptr);

    Data d{};

    d.loop = pw_main_loop_new(nullptr);

    auto props = pw_properties_new(
        PW_KEY_MEDIA_TYPE, "Audio",
        PW_KEY_MEDIA_CATEGORY, "Filter",
        PW_KEY_MEDIA_ROLE, "DSP",
        PW_KEY_NODE_NAME, "Mini Auto Gain",
        PW_KEY_NODE_DESCRIPTION, "Mini Auto Gain",
        PW_KEY_NODE_AUTOCONNECT, "true",
        PW_KEY_NODE_LINK_GROUP, "mini-autogain",
        PW_KEY_FILTER_SMART, "true",
        nullptr
    );

    d.filter = pw_filter_new_simple(
        pw_main_loop_get_loop(d.loop),
        "Mini Auto Gain",
        props,
        &events,
        &d
    );

    if (!d.filter) {
        std::cerr << "Impossible de créer le filtre\n";
        return 1;
    }

    d.in_l = pw_filter_add_port(
        d.filter,
        PW_DIRECTION_INPUT,
        0,
        0,
        pw_properties_new(
            PW_KEY_FORMAT_DSP, "32 bit float mono audio",
            PW_KEY_PORT_NAME, "input_FL",
            nullptr),
        nullptr, 0);

    d.in_r = pw_filter_add_port(
        d.filter,
        PW_DIRECTION_INPUT,
        0,
        0,
        pw_properties_new(
            PW_KEY_FORMAT_DSP, "32 bit float mono audio",
            PW_KEY_PORT_NAME, "input_FR",
            nullptr),
        nullptr, 0);

    d.out_l = pw_filter_add_port(
        d.filter,
        PW_DIRECTION_OUTPUT,
        0,
        0,
        pw_properties_new(
            PW_KEY_FORMAT_DSP, "32 bit float mono audio",
            PW_KEY_PORT_NAME, "output_FL",
            nullptr),
        nullptr, 0);

    d.out_r = pw_filter_add_port(
        d.filter,
        PW_DIRECTION_OUTPUT,
        0,
        0,
        pw_properties_new(
            PW_KEY_FORMAT_DSP, "32 bit float mono audio",
            PW_KEY_PORT_NAME, "output_FR",
            nullptr),
        nullptr, 0);

    int result = pw_filter_connect(
        d.filter,
        PW_FILTER_FLAG_RT_PROCESS,
        nullptr,
        0
    );

    if (result < 0) {
        std::cerr << "Erreur connexion : " << result << "\n";
        return 1;
    }

    std::cout << "Mini Auto Gain actif\n";

    pw_main_loop_run(d.loop);

    pw_filter_destroy(d.filter);
    pw_main_loop_destroy(d.loop);
    pw_deinit();

    return 0;
}
```

```cpp
struct Data {
    pw_main_loop *loop;
    pw_stream *stream;
};

static void on_process(void *userdata)
{

    pw_buffer *b = pw_stream_dequeue_buffer(d->stream);
    if (!b)
        return;
```

```cpp
#include <pipewire/pipewire.h>
#include <spa/param/audio/format-utils.h>
#include <spa/utils/ringbuffer.h>

#include <algorithm>
#include <cstring>
#include <iostream>

static constexpr uint32_t RATE = 48000;
static constexpr uint32_t CHANNELS = 2;
static constexpr uint32_t RING_FRAMES = 32768;
static constexpr uint32_t STRIDE = sizeof(float) * CHANNELS;

struct Data {
    pw_main_loop *loop;
    pw_stream *input;
    pw_stream *output;

    spa_ringbuffer ring;
    float ring_data[RING_FRAMES * CHANNELS];
};

static void on_input(void *userdata)
{
    auto *d = static_cast<Data *>(userdata);

    pw_buffer *b = pw_stream_dequeue_buffer(d->input);
    if (!b)
        return;

    spa_buffer *buf = b->buffer;
    auto *chunk = buf->datas[0].chunk;
    auto *src = static_cast<uint8_t *>(buf->datas[0].data);

    if (!src || !chunk) {
        pw_stream_queue_buffer(d->input, b);
        return;
    }

    uint32_t frames = chunk->size / STRIDE;
    uint32_t index = 0;

    int32_t filled = spa_ringbuffer_get_write_index(
        &d->ring, &index);

    if (filled >= 0) {
        uint32_t space = RING_FRAMES - filled;
        uint32_t count = std::min(frames, space);

        if (count > 0) {
            spa_ringbuffer_write_data(
                &d->ring,
                d->ring_data,
                sizeof(d->ring_data),
                (index % RING_FRAMES) * STRIDE,
                src + chunk->offset,
                count * STRIDE);

            spa_ringbuffer_write_update(
                &d->ring, index + count);
        }
    }

    pw_stream_queue_buffer(d->input, b);
}

static void on_output(void *userdata)
{
    auto *d = static_cast<Data *>(userdata);

    pw_buffer *b = pw_stream_dequeue_buffer(d->output);
    if (!b)
        return;

    spa_buffer *buf = b->buffer;
    auto *dst = static_cast<uint8_t *>(buf->datas[0].data);

    if (!dst) {
        pw_stream_queue_buffer(d->output, b);
        return;
    }

    uint32_t frames =
        buf->datas[0].maxsize / STRIDE;

    if (b->requested)
        frames = std::min(
            frames,
            static_cast<uint32_t>(b->requested));

    uint32_t index = 0;

    int32_t available =
        spa_ringbuffer_get_read_index(
            &d->ring, &index);

    uint32_t count = 0;

    if (available > 0)
        count = std::min(
            static_cast<uint32_t>(available),
            frames);

    if (count > 0) {
        spa_ringbuffer_read_data(
            &d->ring,
            d->ring_data,
            sizeof(d->ring_data),
            (index % RING_FRAMES) * STRIDE,
            dst,
            count * STRIDE);

        spa_ringbuffer_read_update(
            &d->ring, index + count);
    }

    // Silence si le buffer manque de données
    if (count < frames) {
        std::memset(
            dst + count * STRIDE,
            0,
            (frames - count) * STRIDE);
    }

    // Pour l'instant : gain ×1

    buf->datas[0].chunk->offset = 0;
    buf->datas[0].chunk->stride = STRIDE;
    buf->datas[0].chunk->size = frames * STRIDE;

    pw_stream_queue_buffer(d->output, b);
}

static const pw_stream_events input_events = {
    PW_VERSION_STREAM_EVENTS,
    .process = on_input
};

    PW_VERSION_STREAM_EVENTS,
    .process = on_output
};

{

    Data d{};
    spa_ringbuffer_init(&d.ring);

    d.loop = pw_main_loop_new(nullptr);

    d.input = pw_stream_new_simple(

            PW_KEY_MEDIA_TYPE, "Audio",
            PW_KEY_MEDIA_CATEGORY, "Playback",
            PW_KEY_MEDIA_ROLE, "DSP",
            PW_KEY_MEDIA_CLASS, "Audio/Sink",
            PW_KEY_NODE_NAME, "Mini Auto Gain",
            PW_KEY_NODE_DESCRIPTION, "Mini Auto Gain",
            PW_KEY_NODE_LINK_GROUP, "mini-autogain",
            "filter.smart", "true",
            "filter.smart.name", "Mini Auto Gain",
            nullptr),
        &input_events,
        &d);

    // ----- Sortie : Stream/Output/Audio -----

    d.output = pw_stream_new_simple(
        pw_main_loop_get_loop(d.loop),
        "Mini Auto Gain Output",
        pw_properties_new(
            PW_KEY_MEDIA_TYPE, "Audio",
            PW_KEY_MEDIA_CATEGORY, "Playback",
            PW_KEY_MEDIA_ROLE, "DSP",
            PW_KEY_MEDIA_CLASS, "Stream/Output/Audio",
            PW_KEY_NODE_NAME, "Mini Auto Gain Output",
            PW_KEY_NODE_LINK_GROUP, "mini-autogain",
            PW_KEY_NODE_PASSIVE, "true",
            "stream.dont-remix", "true",
            nullptr),
        &output_events,

    spa_audio_info_raw audio{};

    int r1 = pw_stream_connect(
        d.input,
        PW_DIRECTION_INPUT,
        PW_ID_ANY,
        static_cast<pw_stream_flags>(
            PW_STREAM_FLAG_AUTOCONNECT |
            PW_STREAM_FLAG_MAP_BUFFERS |
            PW_STREAM_FLAG_RT_PROCESS),
        &in_params,
        1);

    int r2 = pw_stream_connect(
        d.output,
        PW_DIRECTION_OUTPUT,
        PW_ID_ANY,
        static_cast<pw_stream_flags>(
            PW_STREAM_FLAG_AUTOCONNECT |
            PW_STREAM_FLAG_MAP_BUFFERS |
            PW_STREAM_FLAG_RT_PROCESS),
        &out_params,
        1);

    if (r1 < 0 || r2 < 0) {
        std::cerr << "Erreur PipeWire : "
                  << r1 << " / " << r2 << "\n";
        return 1;
    }

    std::cout << "Mini Auto Gain actif - gain x1\n";

    pw_main_loop_run(d.loop);

    pw_stream_destroy(d.output);
    pw_stream_destroy(d.input);
    pw_main_loop_destroy(d.loop);

    pw_deinit();
    return 0;
}
```

```cpp
// Cherche le peak du buffer
float peak = 0.0f;

float *samples = reinterpret_cast<float *>(dst);

for (uint32_t i = 0; i < frames * CHANNELS; ++i)
    peak = std::max(peak, std::fabs(samples[i]));

// Gain souhaité
float target_gain = MAX_GAIN;

if (peak > 0.000001f)
    target_gain = std::min(MAX_GAIN, TARGET / peak);

// Peak trop fort : baisse immédiate
if (target_gain < d->gain) {
    d->gain = target_gain;
}
// Signal plus faible : remontée linéaire sur 5 s
else {
    float step = (target_gain - d->gain)
               * frames / (RELEASE * RATE);

    d->gain += step;

    if (d->gain > target_gain)
        d->gain = target_gain;
}

// Applique UN seul gain aux deux canaux
for (uint32_t i = 0; i < frames * CHANNELS; ++i)
    samples[i] *= d->gain;
```
